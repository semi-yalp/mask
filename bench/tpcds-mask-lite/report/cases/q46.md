## 原始 SQL

```sql
SELECT c_last_name,
       c_first_name,
       ca_city,
       bought_city,
       ss_ticket_number,
       amt,
       profit
FROM
    (SELECT ss_ticket_number,
            ss_customer_sk,
            ca_city bought_city,
            sum(ss_coupon_amt) amt,
            sum(ss_net_profit) profit
     FROM store_sales,
          date_dim,
          store,
          household_demographics,
          customer_address
     WHERE store_sales.ss_sold_date_sk = date_dim.d_date_sk
       AND store_sales.ss_store_sk = store.s_store_sk
       AND store_sales.ss_hdemo_sk = household_demographics.hd_demo_sk
       AND store_sales.ss_addr_sk = customer_address.ca_address_sk
       AND (household_demographics.hd_dep_count = 4
         OR household_demographics.hd_vehicle_count= 3)
       AND date_dim.d_dow IN (6,
                              0)
       AND date_dim.d_year IN (1999,
                               1999+1,
                               1999+2)
       AND store.s_city IN ('Fairview',
                            'Midway')
     GROUP BY ss_ticket_number,
              ss_customer_sk,
              ss_addr_sk,
              ca_city) dn,
    customer,
    customer_address current_addr
WHERE ss_customer_sk = c_customer_sk
  AND customer.c_current_addr_sk = current_addr.ca_address_sk
  AND current_addr.ca_city <> bought_city
ORDER BY c_last_name NULLS FIRST,
         c_first_name NULLS FIRST,
         ca_city NULLS FIRST,
         bought_city NULLS FIRST,
         ss_ticket_number NULLS FIRST
    LIMIT 100
```

## 改写产物 — 仅脱敏

命中脱敏策略，输出 UDF 包装层：

```sql
SELECT mask_name(r.c_last_name) AS c_last_name, mask_name(r.c_first_name) AS c_first_name, r.ca_city, r.bought_city, r.ss_ticket_number, r.amt, r.profit FROM (
SELECT c_last_name, c_first_name, ca_city, bought_city, ss_ticket_number, amt, profit
FROM (SELECT ss_ticket_number, ss_customer_sk, ca_city AS bought_city, SUM(ss_coupon_amt) AS amt, SUM(ss_net_profit) AS profit
FROM store_sales,
date_dim,
store,
household_demographics,
customer_address
WHERE store_sales.ss_sold_date_sk = date_dim.d_date_sk AND store_sales.ss_store_sk = store.s_store_sk AND store_sales.ss_hdemo_sk = household_demographics.hd_demo_sk AND store_sales.ss_addr_sk = customer_address.ca_address_sk AND (household_demographics.hd_dep_count = 4 OR household_demographics.hd_vehicle_count = 3) AND date_dim.d_dow IN (6, 0) AND date_dim.d_year IN (1999, 1999 + 1, 1999 + 2) AND store.s_city IN ('Fairview', 'Midway')
GROUP BY ss_ticket_number, ss_customer_sk, ss_addr_sk, ca_city) AS dn,
customer,
customer_address AS current_addr
WHERE ss_customer_sk = c_customer_sk AND customer.c_current_addr_sk = current_addr.ca_address_sk AND current_addr.ca_city <> bought_city
ORDER BY c_last_name NULLS FIRST, c_first_name NULLS FIRST, ca_city NULLS FIRST, bought_city NULLS FIRST, ss_ticket_number NULLS FIRST
FETCH NEXT 100 ROWS ONLY
) AS r;
```

## 改写产物 — 仅行过滤

未命中脱敏策略，注入了行过滤谓词：

```sql
SELECT c_last_name, c_first_name, ca_city, bought_city, ss_ticket_number, amt, profit
FROM (SELECT ss_ticket_number, ss_customer_sk, ca_city AS bought_city, SUM(ss_coupon_amt) AS amt, SUM(ss_net_profit) AS profit
FROM store_sales,
(SELECT *
FROM date_dim
WHERE d_year <= 2002) AS date_dim,
store,
household_demographics,
(SELECT *
FROM customer_address
WHERE ca_country = 'United States') AS customer_address

(SELECT *
FROM customer
WHERE c_birth_year >= 1930) AS customer,
(SELECT *
FROM customer_address
WHERE ca_country = 'United States') AS current_addr
```
